\begin{abstract}
Consider the explicit sparse Fock series
\[
 F(z)=\sum_{j\ge1}\frac{z^{\lfloor j^{3/2}\rfloor}}{\sqrt{\lfloor j^{3/2}\rfloor!}}.
\]
For every nonempty open set $U\subset\C$, the derivative $F^{(n)}$ has a zero in $U$ for every sufficiently large $n$.  On each fixed annulus, the zeros have a quantitative two-term description: every zero lies in a model disk, the relevant model disks are eventually pairwise disjoint, each contains exactly one simple zero, and the covering radius is $O(n^{-1/6})$.  The construction is deterministic.  The accompanying Lean~4/Mathlib development verifies every theorem, proposition, and corollary stated in this note and formalizes the explicit inequalities used in their proofs.
\end{abstract}
\maketitle

\section{Statement and context}\label{sec:intro}

The literal formulation of Erd\H{o}s Problem~906 allows a polynomial, since all sufficiently high derivatives then vanish identically.  The nontrivial form asks for a transcendental entire function.  For such a function $g$, write $\Zer(g)$ for its zero set and consider the cofinite zero-hitting property
\begin{equation}\label{eq:cofinite}
 \forall U\subset\C\text{ nonempty open},\qquad
 \exists N(U)\quad\forall n\ge N(U):\quad \Zer(g^{(n)})\cap U\ne\varnothing.
\end{equation}
This has the same quantifier content as the usual formulation using every strictly increasing sequence of derivative orders.  If \eqref{eq:cofinite} fails for some $U$, the infinitely many exceptional orders contain a strictly increasing sequence all of whose selected derivatives miss $U$.  Conversely, if \eqref{eq:cofinite} holds, then every strictly increasing sequence $n_k$ eventually satisfies $n_k\ge N(U)$, so all sufficiently late selected derivatives have a zero in $U$.

The label ``Erd\H{o}s Problem~906'' follows the numbering of the contemporary Erd\H{o}s Problems database; it is not a numbering from Erd\H{o}s's original papers.  That database traces the problem to a 1956 Hungarian source and to Erd\H{o}s's 1982 survey \cite{Bloom906}.  In the 1982 survey, Erd\H{o}s states that the two existence questions discussed there had been proved more than ten years earlier \cite[p.~72]{Erdos82}.  The survey passage does not identify a specific theorem establishing item~(i), the cofinite statement used here.  The Erd\H{o}s Problems database notes that the surrounding context suggests work of Barth and Schneider, but the directly checked 1972 Barth--Schneider theorem \cite{BS72} is a related interpolation result: for prescribed discrete sets $S_k$, one may choose derivative orders $n_k$ and a transcendental entire function whose $n_k$-th derivative vanishes on $S_k$.  It does not contain the cofinite quantifier in \eqref{eq:cofinite}.  Gethner's final-set theory likewise concerns points approached by zeros of infinitely many derivatives, again a weaker quantifier than \eqref{eq:cofinite} \cite{Gethner85}.  Hou gave a probabilistic bounded-coefficient Fock-series construction satisfying \eqref{eq:cofinite}, together with a Lean formalization of the main theorem and growth estimate \cite{Hou26}.  The present construction is explicit and sparse, and its derivative zeros admit a quantitative localization theory.  I make no historical-priority claim beyond these stated features.

Put
\begin{equation}\label{eq:series}
 \nu_j=\lfloor j^{3/2}\rfloor,\qquad
 q_j=\nu_{j+1}-\nu_j,\qquad
 F(z)=\sum_{j=1}^{\infty}\frac{z^{\nu_j}}{\sqrt{\nu_j!}}.
\end{equation}
For $0<a\le b$, let
\[
 \Aab=\{z\in\C:a\le |z|\le b\}.
\]
For an active index $j$, meaning $\nu_j\ge n$, define
\begin{equation}\label{eq:weights}
 m_j=\nu_j-n,\qquad
 A_j=\frac{\sqrt{\nu_j!}}{m_j!},\qquad
 T_j(z)=A_jz^{m_j},\qquad
 \rho_j=\left(\frac{A_j}{A_{j+1}}\right)^{1/q_j}.
\end{equation}
Thus $|T_j|=|T_{j+1}|$ on $|z|=\rho_j$.  The nonzero zeros of the two-term model $T_j+T_{j+1}$ are
\begin{equation}\label{eq:centers}
 z_{j,\ell}=\rho_j\exp\!\left(\frac{(2\ell+1)\pi i}{q_j}\right),
 \qquad 0\le\ell<q_j.
\end{equation}
For $c>0$, write
\begin{equation}\label{eq:modeldisk}
 \mathcal D_{j,\ell}(c)=\overline D\!\left(z_{j,\ell},\frac{c\rho_j}{q_j}\right).
\end{equation}

\begin{theorem}[Cofinite derivative zeros]\label{thm:cofinite}
For every nonempty open set $U\subset\C$, there is $N$ such that for every $n\ge N$, the derivative $F^{(n)}$ has a zero in $U$.
\end{theorem}

\begin{theorem}[Quantitative annular covering]\label{thm:covering}
Let $0<a\le b$.  There are constants $C>0$ and $N$ such that, for every $n\ge N$ and every $w\in\Aab$, there is a zero $z$ of $F^{(n)}$ satisfying
\begin{equation}\label{eq:coveringrate}
 |z-w|\le Cn^{-1/6}.
\end{equation}
\end{theorem}

\begin{proposition}[One simple zero in each model disk]\label{prop:modeldisk}
Let $0<a\le b$.  There exist $c\in(0,1]$ and $N$ such that, for every $n\ge N$, every active $j$ with $a\le\rho_j\le b$, and every $0\le\ell<q_j$, the disk $\mathcal D_{j,\ell}(c)$ contains exactly one zero of $F^{(n)}$, and that zero is simple.
\end{proposition}

\begin{proposition}[Exhaustion of annular zeros]\label{prop:exclusion}
Let $0<a\le b$ and $c>0$.  For all sufficiently large $n$, every zero $z$ of $F^{(n)}$ with $a\le |z|\le b$ belongs to a disk $\mathcal D_{j,\ell}(c)$ with
\[
 \frac a2\le\rho_j\le2b,
 \qquad 0\le\ell<q_j.
\]
\end{proposition}

\begin{corollary}[Eventual annular simplicity]\label{cor:simplicity}
Let $0<a\le b$.  For all sufficiently large $n$, every zero of $F^{(n)}$ in $\Aab$ is simple.
\end{corollary}

\begin{proposition}[Pairwise disjoint model disks]\label{prop:disjoint}
Let $0<a\le b$ and $0<c\le1$.  For all sufficiently large $n$, the disks $\mathcal D_{j,\ell}(c)$ with active $j$, $a\le\rho_j\le b$, and $0\le\ell<q_j$ are pairwise disjoint.
\end{proposition}

\begin{proposition}[Finite model-disk zero count]\label{prop:count}
Let $0<a\le b$.  There exist $c\in(0,1]$ and $N$ such that, for every $n\ge N$ and every finite set $I$ of pairs $(j,\ell)$ with active $j$, $a\le\rho_j\le b$, and $0\le\ell<q_j$, the union of the disks indexed by $I$ contains exactly $|I|$ distinct zeros of $F^{(n)}$, all simple.
\end{proposition}

The rest of the paper proves these statements in the same structural order as the formal development.  Results for other support exponents, the endpoint $p=4/3$, sectorial asymptotics, limiting zero measures, and exact order and type are not part of this submission.

\section{Support, derivative weights, and crossing radii}\label{sec:weights}

The support has one relevant arithmetic scale.  By the mean-value theorem and the floor error,
\begin{equation}\label{eq:gap}
 \frac32\sqrt j-1\le q_j\le \frac32(\sqrt j+1)+1\qquad(j\ge1).
\end{equation}
Hence, whenever $\nu_j=n+O(\sqrt n)$,
\begin{equation}\label{eq:qscale}
 j\asymp n^{2/3},\qquad q_j\asymp n^{1/3}.
\end{equation}
In particular, if $j_0=\min\{j:\nu_j\ge n\}$, then $m_{j_0}=O(n^{1/3})$.

The series \eqref{eq:series} is entire.  A uniform majorant, proved in the formal development by Cauchy--Schwarz, is
\begin{equation}\label{eq:growthupper}
 |F(z)|\le \sqrt{1+|z|^2}\,\exp\!\left(\frac{1+|z|^2}{2}\right).
\end{equation}
Termwise differentiation is therefore legitimate, and
\begin{equation}\label{eq:derivative}
 F^{(n)}(z)=\sum_{\nu_j\ge n} A_jz^{m_j}.
\end{equation}
The support is infinite, so $F$ is not a polynomial.  The formal proof of transcendence is slightly stronger than this observation: a sufficiently high derivative of a polynomial vanishes identically, whereas the corresponding derivative of $F$ does not.

For integers $k\ge n$ and $r>0$, set
\begin{equation}\label{eq:Phi}
 \Phi_{n,r}(k)=\frac12\log(k!)-\log((k-n)!)+(k-n)\log r.
\end{equation}
Its first difference is
\begin{equation}\label{eq:difference}
 d_{n,r}(k)=\Phi_{n,r}(k+1)-\Phi_{n,r}(k)
 =\log r+\frac12\log(k+1)-\log(k+1-n).
\end{equation}
Writing $m=k-n$ gives
\begin{equation}\label{eq:curvature}
 d_{n,r}(k)-d_{n,r}(k+1)
 =\log\!\left(1+\frac1{m+1}\right)
 -\frac12\log\!\left(1+\frac1{n+m+1}\right)>0.
\end{equation}
Thus $\Phi_{n,r}$ is strictly discretely concave.  Uniformly for $m=O(\sqrt n)$, the drop in \eqref{eq:curvature} is comparable to $n^{-1/2}$.

The crossing radius has the exact representation
\begin{equation}\label{eq:rhoexact}
 \log\rho_j=\frac1{q_j}\sum_{s=1}^{q_j}
 \left\{\log(m_j+s)-\frac12\log(n+m_j+s)\right\}.
\end{equation}
The summand is increasing in $s$, hence the crossing radii are strictly increasing.  The same averaging formula gives the explicit squeeze
\begin{equation}\label{eq:rhosqueeze}
 \frac{m_j+1}{\sqrt{n+m_j+1}}\le \rho_j\le\frac{m_j+q_j}{\sqrt{n+m_j+q_j}}.
\end{equation}
In the saddle range $m_j=O(\sqrt n)$, \eqref{eq:rhosqueeze} and \eqref{eq:qscale} give
\begin{equation}\label{eq:rhoapprox}
 \rho_j=\frac{m_j}{\sqrt n}+O(n^{-1/6}).
\end{equation}
On every fixed compact annulus, consecutive relevant crossing radii satisfy
\begin{equation}\label{eq:rhospace}
 c_{a,b}n^{-1/6}\le \rho_{j+1}-\rho_j\le C_{a,b}n^{-1/6}
\end{equation}
for all sufficiently large $n$, after enlarging the annulus by a fixed factor.  The radial mesh is therefore $n^{-1/6}$, whereas the angular mesh on a crossing circle is $q_j^{-1}=O(n^{-1/3})$.

\section{Tail domination and local zero transfer}\label{sec:local}

Fix a compact saddle range and a relevant crossing $\rho=\rho_j$.  Put
\[
 m=m_j,\qquad q=q_j,\qquad Q=\min(q_{j-1},q_{j+1}).
\]
For large $n$ the index $j$ is at least $2$.  The curvature estimate provides a positive number $c$ such that the full-weight slope at the left end of the block is at least $c$ and at the right end is at most $-c$.  In the present $p=3/2$ situation,
\begin{equation}\label{eq:cscale}
 c\asymp n^{-1/6},\qquad Q\asymp n^{1/3}.
\end{equation}
Let $0\le\gamma<c$ and write $|z|=\rho\ee^\sigma$ with $|\sigma|\le\gamma$.  The formal tail estimate is
\begin{equation}\label{eq:tail_exact}
 \frac{\displaystyle\sum_{h\ne j,j+1}|T_h(z)|}{|T_j(z)|}
 \le
 \ee^{-(c-\gamma)Q}
 \left((m+1)+\frac{\ee^{q\gamma}}{1-\ee^{-(c-\gamma)}}\right).
\end{equation}
This is the estimate used below; in particular, no geometric-series simplification of the left tail is needed.  If $\gamma=C_0/q$ with fixed $C_0>0$, then $q\gamma=O(1)$, $(c-\gamma)Q\asymp n^{1/6}$, and $m=O(\sqrt n)$.  Thus the right-hand side of \eqref{eq:tail_exact} tends to zero despite the polynomial factor.  The explicit numerical lemmas in the Lean development choose a threshold $N$ at which it is smaller than each constant required in the local arguments.

We now prove Proposition~\ref{prop:modeldisk}.  Let $z_0=z_{j,\ell}$ and $q=q_j$.  In logarithmic coordinates $z=z_0\ee^w$, divide by the nonvanishing principal monomial.  The derivative takes the form
\begin{equation}\label{eq:logmodel}
 \Psi(w)=1-\ee^{qw}+E(w).
\end{equation}
Choose once and for all $0<c_0\le1/16$ and put $\zeta=c_0/q$.  Applying \eqref{eq:tail_exact} on $|w|\le\zeta$ gives $|E(w)|\le K_n$, with $K_n\to0$.  At $w=0$, $|\Psi(0)|\le K_n$.  On $|w|=\zeta$, the elementary exponential remainder estimate gives
\begin{equation}\label{eq:boundarymodel}
 |1-\ee^{qw}|\ge c_0/2.
\end{equation}
For large $n$, $K_n<c_0/4$, so $|\Psi|>K_n$ on the boundary.  If $\Psi$ had no zero in the closed disk, the maximum-modulus principle applied to $1/\Psi$ would contradict the smaller value of $|\Psi|$ at the centre.  Hence there is a zero $z$ with
\begin{equation}\label{eq:existdist}
 |z-z_0|\le 2\rho\zeta=\frac{2c_0\rho}{q}.
\end{equation}
This is the minimum-modulus step formalized in \texttt{TwoTerm.lean}.

For uniqueness and simplicity, set $t=z/z_0$ and normalize again:
\begin{equation}\label{eq:tmodel}
 G(t)=1-t^q+R(t).
\end{equation}
Take $s=2c_0/q$, so $qs=2c_0\le1/8$, and use the tail estimate on the doubled disk $|t-1|\le2s$.  The logarithmic radial shift there is at most $4s$, so the same tail theorem applies after increasing the harmless fixed transition constant.  After replacing $K_n$ by this larger tail bound, we have $|R|\le K_n$ on $|t-1|\le2s$, and Cauchy's estimate gives
\begin{equation}\label{eq:Rprime}
 |R'(t)|\le K_n/s\qquad(|t-1|\le s).
\end{equation}
On this disk, $|t^{q-1}-1|\le1/7$ for the fixed choice above, and therefore
\begin{equation}\label{eq:powdiff}
 |t_1^q-t_2^q|\ge\frac{6q}{7}|t_1-t_2|
 \qquad(|t_i-1|\le s).
\end{equation}
For large $n$, $K_n<qs/2$.  If $G(t_1)=G(t_2)=0$, \eqref{eq:Rprime} and \eqref{eq:powdiff} force $t_1=t_2$.  At the zero,
\[
 |qt^{q-1}|\ge 6q/7>K_n/s,
\]
so $G'(t)\ne0$.  The zero from \eqref{eq:existdist} lies in this uniqueness disk, proving Proposition~\ref{prop:modeldisk} with the common disk constant $c=2c_0$ after increasing the threshold $N$ if necessary.

\section{Exhaustion, disjointness, and covering}\label{sec:global}

Fix $r$ in a compact annulus.  Because the crossing radii are strictly increasing, for all large $n$ there is a crossing index $j$ with
\begin{equation}\label{eq:bracket}
 \rho_j\le r<\rho_{j+1}.
\end{equation}
Put
\[
 \sigma=\log(r/\rho_j)\ge0,
 \qquad
 \tau=\log(\rho_{j+1}/r)\ge0.
\]
The formal exclusion argument has exactly three cases.  For a fixed sufficiently small $c_0>0$:
\begin{enumerate}
 \item if $\sigma\le c_0/q_j$, the two-term transition estimate at $\rho_j$ puts every zero on $|z|=r$ in a model disk attached to $\rho_j$;
 \item if $\tau\le c_0/q_{j+1}$, the same argument at $\rho_{j+1}$ puts every zero in a model disk attached to $\rho_{j+1}$;
 \item if both inequalities fail, the middle term $T_{j+1}$ is larger than the sum of all the remaining terms, so $F^{(n)}$ has no zero on that circle.
\end{enumerate}
The third case follows from the same concavity and tail estimate as above, now applied on both sides of the dominant term.  In the first two cases, the transition width is eventually below $1/2$ in logarithmic radius; hence a zero with $a\le|z|\le b$ can only be attached to a crossing radius in $[a/2,2b]$.  This proves Proposition~\ref{prop:exclusion}.

Apply Proposition~\ref{prop:modeldisk} on $A_{a/2,2b}$ and use its disk constant in Proposition~\ref{prop:exclusion}.  Every annular zero lies in a disk whose unique zero is simple.  Corollary~\ref{cor:simplicity} follows.

For Proposition~\ref{prop:disjoint}, two distinct model centres on the same crossing circle satisfy
\begin{equation}\label{eq:angularsep}
 |z_{j,\ell}-z_{j,\ell'}|\ge\frac{4\rho_j}{q_j}
 \qquad(\ell\ne\ell')
\end{equation}
for all sufficiently large relevant $j$.  Hence disks with $c\le1$ on the same circle are disjoint.  On different circles, \eqref{eq:rhospace} gives radial separation of order $n^{-1/6}$, whereas the disk radii are $O(q_j^{-1})=O(n^{-1/3})$.  Thus the whole admissible family is eventually pairwise disjoint.  Proposition~\ref{prop:count} then follows by combining disjointness with the one-simple-zero statement in each disk.

We finally prove Theorem~\ref{thm:covering}.  Given $w\in\Aab$, set $r=|w|$ and choose the crossing index from \eqref{eq:bracket}.  The upper bound in \eqref{eq:rhospace} gives
\begin{equation}\label{eq:radialcover}
 |r-\rho_j|\le C_{a,b}n^{-1/6}.
\end{equation}
Choose the nearest model angle on $|z|=\rho_j$.  Its angular displacement from $w$ is $O(q_j^{-1})=O(n^{-1/3})$.  Proposition~\ref{prop:modeldisk}, applied on a fixed enlargement of the annulus, supplies a true zero within another $O(\rho_j/q_j)=O(n^{-1/3})$ of that model point.  The radial error \eqref{eq:radialcover} dominates, proving \eqref{eq:coveringrate}.

Now let $U\subset\C$ be nonempty and open.  Choose a nonzero $w\in U$ and a closed disk $\overline D(w,\delta)\subset U$ with $\delta<|w|/2$.  This disk is contained in a fixed annulus away from the origin.  For all sufficiently large $n$, the covering radius in Theorem~\ref{thm:covering} is smaller than $\delta$, so $F^{(n)}$ has a zero in $U$.  This proves Theorem~\ref{thm:cofinite}.

\begin{remark}[The origin]\label{rem:origin}
Let $j_0=\min\{j:\nu_j\ge n\}$.  The multiplicity of the zero of $F^{(n)}$ at the origin is exactly $\nu_{j_0}-n$.  This can be large, so the simplicity statement is necessarily restricted to annuli bounded away from the origin.
\end{remark}
